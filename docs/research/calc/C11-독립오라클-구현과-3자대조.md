### 2.1 `jd.js` (Meeus Ch.7)

```js
/** Meeus (7.1). D는 소수 포함 일. gregorian=false 이면 율리우스력. */
function calendarToJD(Y, M, D, gregorian) {
  if (gregorian === undefined) {
    gregorian = (Y > 1582) || (Y === 1582 && (M > 10 || (M === 10 && D >= 15)));
  }
  let y = Y, m = M;
  if (m <= 2) { y -= 1; m += 12; }
  let B = 0;
  if (gregorian) { const A = Math.floor(y / 100); B = 2 - A + Math.floor(A / 4); }
  return Math.floor(365.25 * (y + 4716)) + Math.floor(30.6001 * (m + 1)) + D + B - 1524.5;
}

/** 달력 전환 자동 판정 JDN. 1582-10-15 미만은 율리우스력으로 해석. */
function jdnNoonAuto(y, m, d) { return Math.floor(calendarToJD(y, m, d) + 0.5); }

/** Fliegel–Van Flandern 정수 JDN. 그레고리력 전용(proleptic). */
function jdnNoon(y, m, d) {
  const a = Math.floor((14 - m) / 12);
  const yy = y + 4800 - a, mm = m + 12 * a - 3;
  return d + Math.floor((153 * mm + 2) / 5) + 365 * yy
         + Math.floor(yy / 4) - Math.floor(yy / 100) + Math.floor(yy / 400) - 32045;
}
const JD_UNIX_EPOCH = 2440587.5;                       // 1970-01-01T00:00Z
const msToJD = ms => ms / 86400000 + JD_UNIX_EPOCH;
const jdToMs = jd => (jd - JD_UNIX_EPOCH) * 86400000;
```

### 2.2 `sun.js` — VSOP87D 고정밀 경로

```js
function vsopSum(series, tau) {                 // series[i] = [[A,B,C],...], A 단위 1e-8
  let total = 0, p = 1;
  for (let i = 0; i < series.length; i++) {
    let s = 0; const arr = series[i];
    for (let j = 0; j < arr.length; j++) s += arr[j][0] * Math.cos(arr[j][1] + arr[j][2] * tau);
    total += s * p; p *= tau;
  }
  return total * 1e-8;                          // rad (L,B) / AU (R)
}
function earthHelio(jde) {                      // τ = 율리우스 천년
  const tau = (jde - 2451545.0) / 365250.0;
  return { L: mod360(vsopSum(D.L, tau) / DEG), B: vsopSum(D.B, tau) / DEG, R: vsopSum(D.R, tau) };
}
function nutation(jde) {                        // Meeus Ch.22, IAU1980 63항
  const T = (jde - 2451545.0) / 36525.0;
  const d  = 297.85036 + T * (445267.111480 + T * (-0.0019142 + T / 189474.0));
  const m  = 357.52772 + T * (35999.050340 + T * (-0.0001603 - T / 300000.0));
  const mp = 134.96298 + T * (477198.867398 + T * (0.0086972 + T / 56250.0));
  const f  =  93.27191 + T * (483202.017538 + T * (-0.0036825 + T / 327270.0));
  const om = 125.04452 + T * (-1934.136261 + T * (0.0020708 + T / 450000.0));
  const args = [d, m, mp, f, om];
  let dpsi = 0, deps = 0;
  for (let i = 0; i < D.nutArg.length; i++) {
    let a = 0;
    for (let j = 0; j < 5; j++) if (D.nutArg[i][j]) a += D.nutArg[i][j] * args[j];
    const ar = a * DEG, s = D.nutSin[i];
    dpsi += (s[0] + (s[1] ? s[1] * T : 0)) * Math.sin(ar);
    if (i < D.nutCos.length) { const c = D.nutCos[i]; deps += (c[0] + (c[1] ? c[1] * T : 0)) * Math.cos(ar); }
  }
  return { dpsiArcsec: dpsi / 10000.0, depsArcsec: deps / 10000.0 };
}
function sunGeometric(jde) {                    // FK5 보정 (Meeus 25.9)
  const e = earthHelio(jde), T = (jde - 2451545.0) / 36525.0;
  const theta = mod360(e.L + 180), beta = -e.B;
  const lp = theta - 1.397 * T - 0.00031 * T * T;
  return { theta: mod360(theta - 0.09033 / 3600),
           beta: beta + 0.03916 * (cosd(lp) - sind(lp)) / 3600, R: e.R };
}
function sunLongitudeHigh(jde) {                // 겉보기 황경
  const g = sunGeometric(jde), nut = nutation(jde);
  return mod360(g.theta + nut.dpsiArcsec / 3600.0 + (-20.4898 / g.R / 3600.0));
}
/** 균시차 (Meeus 28.3): E = L0 − 0.0057183° − α + Δψ·cos ε.  반환 분 */
function equationOfTimeHigh(jde) {
  const tau = (jde - 2451545.0) / 365250.0;
  const L0 = mod360(280.4664567 + 360007.6982779 * tau + 0.03032028 * tau ** 2
            + tau ** 3 / 49931 - tau ** 4 / 15300 - tau ** 5 / 2000000);
  const eq = sunApparentEquatorial(jde), nut = nutation(jde);
  return norm180(L0 - 0.0057183 - eq.ra + (nut.dpsiArcsec / 3600.0) * cosd(eq.eps)) * 4;
}
```

### 2.3 `deltat.js` — 하이브리드 ΔT (★ 본 문서 최대 수정점)

```js
const OBS = require('./deltat-obs.js');          // [[십진연도, ΔT초], ...] 1657.0~2033.75, 1302점
const OBS_MIN = OBS[0][0], OBS_MAX = OBS[OBS.length - 1][0];

function deltaTObserved(y) {                     // 이분탐색 + 선형보간
  if (y <= OBS_MIN) return OBS[0][1];
  if (y >= OBS_MAX) return OBS[OBS.length - 1][1];
  let lo = 0, hi = OBS.length - 1;
  while (hi - lo > 1) { const m = (lo + hi) >> 1; if (OBS[m][0] <= y) lo = m; else hi = m; }
  const [y0, v0] = OBS[lo], [y1, v1] = OBS[hi];
  return v0 + (v1 - v0) * (y - y0) / (y1 - y0);
}
/** [1657, 2033.75] = USNO 관측/예측. 밖 = E&M 다항식을 경계에서 연속이 되게 평행이동 */
function deltaTHybrid(y) {
  if (y >= OBS_MIN && y <= OBS_MAX) return deltaTObserved(y);
  if (y < OBS_MIN) return deltaTSeconds(y) - (deltaTSeconds(OBS_MIN) - OBS[0][1]);
  return deltaTSeconds(y) - (deltaTSeconds(OBS_MAX) - OBS[OBS.length - 1][1]);
}
```

### 2.4 `solarterms.js` — 이분법 역산

```js
const MEAN_RATE = 0.98564736;                    // °/day

function lonAtUT(jdUT, precise) {                // ★ 근찾기를 UT 축에서 한다 (ΔT 를 내부에서 더함)
  const jde = jdUT + deltaTFromJD(jdUT) / 86400;
  return precise ? S.sunLongitudeHigh(jde) : S.sunLongitudeLow(jde);
}
function bisect(target, lo, hi, precise, maxIter) {
  const g = t => S.norm180(lonAtUT(t, precise) - target);   // 0°/360° 랩어라운드 자동 처리
  let flo = g(lo), fhi = g(hi);
  if (flo === 0) return lo;
  if (fhi === 0) return hi;
  if (flo * fhi > 0) throw new Error(`bracket fail target=${target} …`);
  for (let i = 0; i < (maxIter || 60); i++) {
    const mid = (lo + hi) / 2, fm = g(mid);
    if (fm === 0) return mid;
    if (flo * fm < 0) { hi = mid; fhi = fm; } else { lo = mid; flo = fm; }
  }
  return (lo + hi) / 2;
}
function termJD_UT(y, target, precise) {
  const equinoxGuess = calendarToJD(y, 3, 20.5);
  const off = (target >= 285 ? target - 360 : target) / MEAN_RATE;
  let guess = equinoxGuess + off;
  // ★ 1차 선형 외삽은 J2000 근방에서만 ±3일이다. 세차 때문에 1391년에는 3월 춘분이
  //   3월 11일경이라 ±6일 bracket 이 깨진다(실측: 1391 立春 bracket fail).
  //   → Meeus Ch.27 고정점 반복으로 guess 를 수렴시킨 뒤 좁은 bracket 을 씌운다.
  for (let i = 0; i < 5; i++) {
    const lam = lonAtUT(guess, precise !== false);
    const corr = 58.0 * Math.sin(S.norm180(target - lam) * Math.PI / 180);
    guess += corr;
    if (Math.abs(corr) < 1e-6) break;
  }
  return bisect(target, guess - 2, guess + 2, precise !== false);
}
```

### 6.4 ★ 대운수 규칙 역공학

golden-set `conventions.daewoonNumber` 는 `"한국식 = round(floor(Δ일)/3)"` 이라고 적혀 있는데,
그 식으로는 275건 중 216건(78.5%)만 맞는다. 6개 후보를 동시 채점한 결과:

| 규칙 | 적중 |
|---|---|
| `round(floor(d)/3)` | 216/275 = 78.5% |
| **`max(1, round(floor(d)/3))`** | **275/275 = 100.0%** |
| `round(d/3)` | 194/275 = 70.5% |
| `max(1, round(d/3))` | 243/275 = 88.4% |
| `floor(floor(d)/3) + (나머지2면 +1)` | 216/275 = 78.5% |
| **`max(1, floor(floor(d)/3) + (나머지2면 +1))`** | **275/275 = 100.0%** |

**빠진 것은 `max(1, ·)` 클램프**다. Δ가 3일 미만이면 위 두 식 모두 0을 내는데 데이터는 전부 1이다.
(59건의 불일치가 전부 `Δ<3일 → 자체 0 vs 기대 1` 형태였다.)
`max(1,round(floor(d)/3))` 와 `max(1,한국전통)` 이 이 287케이스에서 동률이라 **둘을 구별하지 못한다** —
구별하려면 `floor(d) mod 3 == 1` 이면서 `floor(d)/3` 의 소수부가 0.33 인 케이스가 필요하다.
(예: floor(d)=10 → round(3.33)=3, 한국전통 floor(3.33)=3 … 동일. floor(d)=7 → round(2.33)=2, 전통 2 … 동일.
실제로 두 식은 `floor(d) mod 3 == 2` 에서만 갈리는데 round(x.67)=올림 = 전통 올림 → **항상 같다.**
⇒ **`round(floor(d)/3)` 와 `floor(d)/3 후 1버림·2올림` 은 수학적으로 동일한 식이다.**)
